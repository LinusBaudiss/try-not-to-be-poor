declare module 'vitest' {
 type RowArgs<T> = T extends readonly unknown[] ? [...T] : [T];
 interface TestAPI {
   (name: string, callback: () => unknown): void;
   each<T>(rows: readonly T[]): (name: string, callback: (...args: RowArgs<T>) => unknown) => void;
 }
 export const describe: (name: string, callback: () => unknown) => void;
 export const it: TestAPI;
 export const expect: any;
 export const beforeEach: (callback: () => unknown) => void;
 export const afterEach: (callback: () => unknown) => void;
 export const vi: any;
}
